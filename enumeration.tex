\section*{Enumeration Inference Example}

\subsection*{Query}
\[
P(\text{Result} = 1\text{-}0 \mid \text{BigECO} = \text{B20-B99},\; \text{WS} = \text{Neg},\; \text{BS} = \text{Pos})
\]

\subsection*{Numerator}
\begin{flalign*}
& \sum_{\text{QT}} \sum_{\text{Bal}} \sum_{\text{CB}} &\\
& \quad P(\text{Naka}) &\\
& \quad \cdot\; P(\text{B20-B99} \mid \text{Naka}) &\\
& \quad \cdot\; P(\text{QT} \mid \text{Naka}, \text{B20-B99}) &\\
& \quad \cdot\; P(\text{Bal} \mid \text{Naka}, \text{B20-B99}) &\\
& \quad \cdot\; P(\text{CB} \mid \text{Naka}, \text{B20-B99}) &\\
& \quad \cdot\; P(\text{Neg}) &\\
& \quad \cdot\; P(\text{Pos}) &\\
& \quad \cdot\; P(1\text{-}0 \mid \text{QT}, \text{Bal}, \text{CB}, \text{Naka}, \text{Neg}, \text{Pos}) &
\end{flalign*}

\subsection*{Denominator}
\begin{flalign*}
& \sum_{R} \sum_{\text{QT}} \sum_{\text{Bal}} \sum_{\text{CB}} &\\
& \quad P(\text{Naka}) &\\
& \quad \cdot\; P(\text{B20-B99} \mid \text{Naka}) &\\
& \quad \cdot\; P(\text{QT} \mid \text{Naka}, \text{B20-B99}) &\\
& \quad \cdot\; P(\text{Bal} \mid \text{Naka}, \text{B20-B99}) &\\
& \quad \cdot\; P(\text{CB} \mid \text{Naka}, \text{B20-B99}) &\\
& \quad \cdot\; P(\text{Neg}) &\\
& \quad \cdot\; P(\text{Pos}) &\\
& \quad \cdot\; P(R \mid \text{QT}, \text{Bal}, \text{CB}, \text{Naka}, \text{Neg}, \text{Pos}) &
\end{flalign*}

\subsection*{Answer}
\[
P(\text{Result}=1\text{-}0 \mid \text{B20-B99}, \text{Neg}, \text{Pos})
= \frac{\text{Numerator}}{\text{Denominator}}
\]

\subsection*{Values from CPDs}

P(\text{QT}=T \mid \text{Naka}, \text{B20-B99}) &= 0.5418 \\
P(\text{Bal}=F \mid \text{Naka}, \text{B20-B99}) &= 0.7525 &
P(\text{Bal}=T \mid \text{Naka}, \text{B20-B99}) &= 0.2475 \\
P(\text{CB}=F \mid \text{Naka}, \text{B20-B99}) &= 0.3679 &
P(\text{CB}=T \mid \text{Naka}, \text{B20-B99}) &= 0.6321

\subsection*{Computation}

Constant terms $P(\text{Naka})$, $P(\text{B20-B99} \mid \text{Naka})$, $P(\text{Neg})$, $P(\text{Pos})$
cancel between numerator and denominator.

For each combo $i$, compute a weight:
\[
w_i = P(\text{QT}_i \mid \text{Naka}, \text{B20-B99})
\times P(\text{Bal}_i \mid \text{Naka}, \text{B20-B99})
\times P(\text{CB}_i \mid \text{Naka}, \text{B20-B99})
\]

\begin{center}
\begin{tabular}{ccc|ccc|c}
QT & Bal & CB & $P(\text{QT})$ & $P(\text{Bal})$ & $P(\text{CB})$ & $w_i$ \\
\hline
F & F & F & 0.4582 & 0.7525 & 0.3679 & 0.1268 \\
F & F & T & 0.4582 & 0.7525 & 0.6321 & 0.2180 \\
F & T & F & 0.4582 & 0.2475 & 0.3679 & 0.0417 \\
F & T & T & 0.4582 & 0.2475 & 0.6321 & 0.0717 \\
T & F & F & 0.5418 & 0.7525 & 0.3679 & 0.1500 \\
T & F & T & 0.5418 & 0.7525 & 0.6321 & 0.2578 \\
T & T & F & 0.5418 & 0.2475 & 0.3679 & 0.0493 \\
T & T & T & 0.5418 & 0.2475 & 0.6321 & 0.0848 \\
\end{tabular}
\end{center}

Since $P(1\text{-}0) + P(0\text{-}1) + P(\frac{1}{2}\text{-}\frac{1}{2}) = 1$ for each row,
the denominator $= \sum_i w_i = 1$.

The numerator is $\sum_i w_i \times P(1\text{-}0 \mid \text{combo}_i)$:

\begin{flalign*}
& = 0.1268 \times 0.3750 &\\
& + 0.2180 \times 0.3913 &\\
& + 0.0417 \times 0.3333 &\\
& + 0.0717 \times 0.4000 &\\
& + 0.1500 \times 0.4000 &\\
& + 0.2578 \times 0.4348 &\\
& + 0.0493 \times 0.5000 &\\
& + 0.0848 \times 0.2500 &
\end{flalign*}

\subsection*{Answer}
\[
P(\text{Result}=1\text{-}0 \mid \text{B20-B99}, \text{Neg}, \text{Pos})
= \frac{\text{Numerator}}{1} = \text{Numerator}
\]
